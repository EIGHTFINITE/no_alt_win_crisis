ap_defender_of_the_galaxy_nomads = {
	possible = {
		custom_tooltip = {
			fail_text = "requires_ascension_perks_3"
			num_ascension_perks > 2
		}
		NOR = {
			has_ethic = ethic_fanatic_xenophobe
			has_civic = civic_inwards_perfection
		}
		custom_tooltip = {
			fail_text = "requires_actor_not_khan"
			NOT = { has_country_flag = heir_of_the_khan_khan }
		}
		OR = {
			NOR = {
				has_ascension_perk = ap_become_the_crisis
				has_ascension_perk = ap_cosmogenesis
				has_ascension_perk = ap_behemoths
				has_ascension_perk = ap_galactic_hyperthermia
			}
			AND = {
				has_ascension_perk = ap_become_the_crisis
				has_crisis_level = crisis_level_5
			}
			AND = {
				has_ascension_perk = ap_cosmogenesis
				has_crisis_level = crisis_cosmogenesis_level_5
			}
			AND = {
				has_ascension_perk = ap_behemoths
				has_crisis_level = crisis_behemoth_level_5
			}
			AND = {
				has_ascension_perk = ap_galactic_hyperthermia
				has_crisis_level = crisis_hyperthermia_level_5
			}
		}
	}
	#custom_tooltip = ap_defender_of_the_galaxy_modifier_desc

	modifier = {
		add_attunement_the_cradle_of_souls = @ascension_perk_attunement1
	}

	on_enabled = {
		activate_crisis_progression = defender_path
	}

	potential = {
		has_nomads_dlc = yes
		NOR = {
			has_defender_of_the_galaxy_ap = yes
			has_country_flag = is_player_crisis
			has_origin = origin_endbringers # Endbringers are technically a crisis as well
			is_unfriendly = yes
		}
	}

	ai_weight = {
		factor = 5
		modifier = {
			factor = 10
			OR = {
				has_crisis = yes
				any_country = {
					relative_power = {
						who = prev
						value > equivalent
					}
					has_country_flag = is_player_crisis
				}
			}
		}
		modifier = {
			factor = 0
			NOR = {
				is_xenophile = yes
				is_galactic_custodian = yes
				has_valid_civic = civic_hive_familiar_face
				any_country = {
					OR = {
						is_level_5_player_crisis = yes
						is_crisis_faction = yes
					}
				}
			}
		}
		modifier = {
			factor = 100
			is_galactic_custodian = yes
		}
		modifier = {
			factor = 0.25
			NOT = {
				any_country = {
					OR = {
						is_level_5_player_crisis = yes
						is_crisis_faction = yes
					}
				}
			}
		}
	}
}

ap_become_the_crisis = {
	potential = {
		has_nemesis = yes
		NOR = {
			has_country_flag = is_player_crisis
			has_defender_of_the_galaxy_ap = yes
			has_origin = origin_slavers
			has_origin = origin_endbringers # No additional crisis for Endbringers
		}
	}

	possible = {
		custom_tooltip = {
			fail_text = "requires_independence"
			is_subject = no
		}
		custom_tooltip = {
			fail_text = "must_not_be_custodian"
			is_galactic_custodian = no
		}
		custom_tooltip = {
			fail_text = "must_not_be_emperor"
			is_galactic_emperor = no
		}
		OR = {
			NOR = {
				has_ascension_perk = ap_cosmogenesis
				has_ascension_perk = ap_behemoths
				has_ascension_perk = ap_galactic_hyperthermia
				has_ascension_perk = ap_defender_of_the_galaxy_nomads
			}
			AND = {
				has_ascension_perk = ap_cosmogenesis
				has_crisis_level = crisis_cosmogenesis_level_5
			}
			AND = {
				has_ascension_perk = ap_behemoths
				has_crisis_level = crisis_behemoth_level_5
			}
			AND = {
				has_ascension_perk = ap_galactic_hyperthermia
				has_crisis_level = crisis_hyperthermia_level_5
			}
			AND = {
				has_ascension_perk = ap_defender_of_the_galaxy_nomads
				has_crisis_level = crisis_defender_level_5
			}
		}
		custom_tooltip = {
			fail_text = "must_finish_behemoths_path"
			OR = {
				is_wilderness_empire = no
				has_ascension_perk = ap_behemoths
			}
		}
		custom_tooltip = {
			fail_text = "must_finish_shroud_forged_situation"
			OR = {
				NOT = { has_origin = origin_shroud_forged }
				NOT = {
					any_situation = {
						is_situation_type = situation_shroud_forged
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = "locked_by_shroud_forged_situation"
			OR = {
				NOT = { has_origin = origin_shroud_forged }
				NOT = { has_country_flag = finish_shroud_forged_liberation_flag }
			}
		}
		if = {
			limit = {
				OR = {
					NOT = { has_origin = origin_shroud_forged }
					NOT = { has_country_flag = shroud_forged_can_become_the_crisis }
				}
			}
			custom_tooltip = {
				fail_text = "requires_ascension_perks_3"
				num_ascension_perks > 2
			}
			NOR = {
				is_xenophile = yes
				is_pacifist = yes
				has_civic = civic_machine_servitor
				has_valid_civic = civic_hive_familiar_face
			}
		}
	}

	on_enabled = {
		hidden_effect = {
			set_ai_personality = became_the_crisis
			country_event = { id = timeline.69 } # Timeline Become the crisis
		}
		activate_crisis_progression = nemesis_path
	}

	modifier = {
		refugee_attraction = -0.50
	}

	ai_weight = {
		factor = 7.5
		modifier = {
			factor = 0
			NOR = {
				is_militarist = yes
				is_xenophobe = yes
				is_gestalt = yes
			}
		}
		modifier = {
			factor = 10
			is_homicidal = yes
		}
		modifier = {
			factor = 5
			any_playable_country = {
				has_defender_of_the_galaxy_ap = yes
			}
			NOT = {
				any_playable_country = {
					has_ascension_perk = ap_become_the_crisis
				}
			}
		}
		modifier = {
			factor = 2
			has_tradition = tr_supremacy_finish
		}
		modifier = {
			factor = 0
			galaxy_percentage < 0.1
		}
		modifier = {
			factor = 0.5
			is_infernal_empire = yes
		}
		modifier = {
			factor = 2
			galaxy_percentage > 0.2
		}
		modifier = {
			factor = 0
			count_playable_country = {
				count > 1
				limit = {
					has_ascension_perk = ap_become_the_crisis
				}
			}
		}
		modifier = {
			factor = 0
			blocks_disruptive_gameplay = yes
		}
	}
}
