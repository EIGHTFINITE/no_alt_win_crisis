		activate_crisis_progression = defender_path

		has_nomads_dlc = yes

			has_country_flag = is_player_crisis

	}

	ai_weight = {
		factor = 5
		modifier = {
			factor = 10

				has_crisis = yes

ap_force_of_will = {
	potential = {
		has_force_of_will_dlc = yes
		NOR = {
			has_country_flag = is_player_crisis
			has_origin = origin_endbringers
			has_ascension_perk = ap_defender_of_the_galaxy
			has_ascension_perk = ap_defender_of_the_galaxy_nomads
		}
	}

	possible = {
		custom_tooltip = {
			fail_text = "requires_ascension_perks_3"
			num_ascension_perks > 2
		}
		custom_tooltip = {
			fail_text = "must_finish_crisis_path"
			OR = {
				has_ambition = no
				AND = {
					has_ascension_perk = ap_defender_of_the_galaxy_nomads
					has_crisis_level = crisis_defender_level_5
				}
				AND = {
					has_ascension_perk = ap_become_the_crisis
					has_crisis_level = crisis_level_5
				}
				AND = {
					has_ascension_perk = ap_cosmogenesis
					has_crisis_level = crisis_cosmogenesis_level_5
				}
				AND = {
					has_ascension_perk = ap_behemoths
					has_crisis_level = crisis_behemoth_level_5
				}
				AND = {
					has_ascension_perk = ap_galactic_hyperthermia
					has_crisis_level = crisis_hyperthermia_level_5
				}
			}
		}
		custom_tooltip = {
			fail_text = requires_fanatic
			is_fanatic = yes
		}
		if = {
			limit = {
				NOT = { exists = ideology }
			}
			custom_tooltip = {
				fail_text = "must_have_good_ideology"
				always = no
			}
		}
		else = {
			custom_tooltip = {
				fail_text = "must_be_ideological_custodian"
				ideology = {
					is_ideology_custodian = root
				}
			}
			custom_tooltip = {
				fail_text = requires_ideology_not_militarist
				ideology = {
					is_militarist = no
				}
			}
			custom_tooltip = {
				fail_text = requires_ideology_not_xenophobe
				ideology = {
					is_xenophobe = no
				}
			}
			custom_tooltip = {
				fail_text = requires_ideology_not_authoritarian
				ideology = {
					is_authoritarian = no
				}
			}
		}
	}

	on_enabled = {
		activate_crisis_progression = force_of_will_path
		set_country_flag = cant_abandon_ideology_forceofwill
		if = {
			limit = { exists = ideology }
			custom_tooltip = ap_force_of_will_tooltip
		}
		if = {
			limit = { has_country_flag = abandoning_ideology }
			custom_tooltip = ap_force_of_will_stop_abandon_tooltip
		}
		if = {
			limit = {
				any_war = {
					war_imposes_ideology_on_root = yes
				}
			}
			custom_tooltip = ap_force_of_will_leave_ideological_war_tooltip
		}
		hidden_effect = {
			if = {
				limit = { has_country_flag = abandoning_ideology }
				stop_abandon_ideology_process = yes
			}
			if = {
				limit = {
					any_war = {
						war_imposes_ideology_on_root = yes
					}
				}
				country_event = {
					id = forceofwill.6140
					days = 3
				}
			}
			every_war = {
				limit = {
					war_imposes_ideology_on_root = yes
				}
				if = {
					limit = {
						defender = { is_same_value = root }
					}
					attacker = {
						country_event = {
							id = forceofwill.6145
							scopes = { from = root }
							days = 1
						}
					}
					set_war_goal = {
						type = wg_humiliation
						target = attacker
					}
				}
				else = {
					defender = {
						country_event = {
							id = forceofwill.6145
							scopes = { from = root }
							days = 1
						}
					}
					set_war_goal = {
						type = wg_humiliation
						target = defender
					}
				}
				remove_war_participant = root
			}
		}
	}

	tradition_swap = {
		name = ap_force_of_will_spiritual
		inherit_effects = yes
		inherit_icon = no
		trigger = {
			ideology? = {
				is_spiritualist = yes
			}
		}
	}

	ai_weight = {
		factor = 7.5
		modifier = {
			factor = 2
			last_ideology_increase > 1.5
		}
		modifier = {
			factor = 3
			ideology? = {
				root = {
					has_same_fanatic_ethic_as_ideology = prev
				}
			}
		}
		modifier = {
			factor = 2
			ideological_strength > 750
		}
		modifier = {
			factor = 5
			ideology? = {
				galactic_strength >= 0.35
				galactic_strength < 0.5
			}
		}
		modifier = {
			factor = 10
			ideology? = {
				galactic_strength >= 0.5
			}
		}
	}
}

			}
