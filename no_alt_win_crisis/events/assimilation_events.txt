# Each type of assimilation uses a separate X00 block of namespace to future proof this.

namespace = assimilation

# A pop amount has been grown, try to transfer it to assimilation target if applicable
# This = Planet scope
# From = Country
# FromFrom = Pop group
# local_pop_amount = variable of how much was grown
carrier_event = {
	id = assimilation.40
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		check_variable = {
			which = local_pop_amount
			value > 0
		}
		fromfrom = {
			species = {
				has_citizenship_type = {
					type = citizenship_assimilation
					country = root.from
				}
				any_galaxy_species = {
					has_species_flag = assimilation_species_of_species@prev
					has_species_flag = assimilation_species_of_empire@root.from
					is_same_species = prev
				}
			}
			pop_group_size > 0
		}
	}

	immediate = {
		fromfrom = {
			save_event_target_as = old_pop_group
			species = {
				save_event_target_as = old_species
			}
		}
		random_galaxy_species = {
			limit = {
				has_species_flag = assimilation_species_of_species@event_target:old_species
				has_species_flag = assimilation_species_of_empire@root.from
				is_same_species = event_target:old_species
			}
			save_event_target_as = target_species
		}
		if = {
			limit = {
				check_variable = {
					which = local_pop_amount
					value > 0
				}
				event_target:old_pop_group = {
					pop_group_size > 0
				}
			}
			create_pop_group = {
				pop_group = event_target:old_pop_group
				species = event_target:target_species
				size = 0
				effect = {
					transfer_pop_amount = {
						source = event_target:old_pop_group
						target = this
						amount = root.local_pop_amount
						growth_category = "GROWTH_CAT_GROWTH"
					}
					apply_post_assimilation_effects = yes
				}
			}
		}
	}
}

# Cyborg Assimilation - Hive Mind
carrier_event = {
	id = assimilation.210
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_owned_species = {
			has_cyborg_assimilation_living_standard = {
				COUNTRY = root.owner
			}
		}
	}

	immediate = {
		inline_script = {
			script = assimilation/run
			TYPE = cyborg
			TARGET_SPECIES = create_assimilation_species
			TRAITS = "
				add_trait = trait_cybernetic
				remove_trait = trait_self_modified
				inline_script = traits/remove_all_psionic_traits
				add_trait = trait_hive_mind
			"
			TRANSFER = assimilate_pop_group_effect
		}

		# grant timed modifiers from cyber hive
		if = {
			limit = {
				owner = {
					OR = {
						has_tradition = tr_cybernetics_finish
						AND = {
							has_origin = origin_nomadic_purger
							overlord? = {
								has_tradition = tr_cybernetics_finish
							}
						}
					}
				}
			}
			# A purge vassal disbands, so the empire-wide node boost goes to its Overlord; the planet boost stays.
			if = {
				limit = {
					owner = {
						has_origin = origin_nomadic_purger
					}
				}
				owner = {
					overlord? = {
						add_modifier = {
							modifier = cyber_hive_node_booster
							days = 90
						}
					}
				}
			}
			else = {
				owner = {
					add_modifier = {
						modifier = cyber_hive_node_booster
						days = 90
					}
				}
			}
			add_modifier = {
				modifier = cyber_hive_planet_booster
				days = 90
			}
		}
	}
}

carrier_event = {
	id = assimilation.415
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_owned_species = {
			is_robotic = yes
			has_citizenship_type = {
				type = citizenship_assimilation
				country = root.owner
			}
		}
	}

	immediate = {
		set_assimilation_counter_variable = yes
		set_update_modifiers_batch = begin			# Freeze modifier calculation for performance

		while = {
			count = assimilation_counter

			random_owned_pop_group = {
				limit = {
					pop_group_size > 0
					NOT = {
						has_pop_group_flag = already_assimilated
					}
					is_robotic_species = yes
					has_citizenship_type = {
						type = citizenship_assimilation
						country = root.owner
					}
				}
				save_event_target_as = old_pop_group
				species = {
					save_event_target_as = old_species
				}
				# Synthetic/Machine Assimilation just turns you into the founder species
				root.owner_species = {
					save_event_target_as = target_species
					if = {
						limit = {
							root.owner = {
								has_modifier = synthetic_dawn
							}
							NOT = {
								has_trait = trait_robot_synthetic_dawn
							}
						}
						create_species = {
							is_mod = yes
							name = this
							plural = this
							class = this
							portrait = this
							traits = this
							can_be_modified = this
							homeworld = this
							namelist = this
							gender = this
							effect = {
								modify_species = {
									add_trait = trait_robot_synthetic_dawn
									effect = {
										save_event_target_as = target_species
									}
								}
							}
						}
					}
				}
				assimilate_pop_group_effect = yes
			}
		}

		finish_assimilation_effect = yes
		if = {
			limit = { has_ascension_perk = ap_become_the_crisis }
			add_resource = { menace = 3 }
		}
		else_if = {
			limit = {
				has_origin = origin_nomadic_purger
				overlord? = { has_ascension_perk = ap_become_the_crisis }
			}
			overlord = {
				add_resource = { menace = 3 }
			}
		}

		set_update_modifiers_batch = end			# Unfreeze modifier calculation for performance
		clear_assimilation_variables = yes
	}
}
